function leGetPositionFrequencies($db, string $dbCol, string $position, int $drawRange, string $gId): array
{
    $q = $db->getQuery(true)
        ->select($db->quoteName($position))
        ->from($db->quoteName($dbCol))
        ->where($db->quoteName('game_id') . ' = :gid')
        ->order($db->quoteName('draw_date') . ' DESC')
        ->setLimit($drawRange);

    $q->bind(':gid', $gId, ParameterType::STRING);
    $db->setQuery($q);

    $col = $db->loadColumn();
    if (!is_array($col)) {
        return [];
    }

    // Strict sanitation — force values into a single digit 0–9 (Pick 3 positions must be 0–9)
    $vals = [];
    foreach ($col as $v) {
        $s = trim((string) $v);

        if ($s === '' || strcasecmp($s, 'n/a') === 0 || strcasecmp($s, 'null') === 0) {
            continue;
        }

        if (preg_match('/^[0-9]$/', $s) === 1) {
            $vals[] = $s;
            continue;
        }

        if (preg_match('/^[0-9]+$/', $s) === 1) {
            $vals[] = substr($s, -1);
            continue;
        }

        if (preg_match('/([0-9])\D*$/', $s, $m) === 1) {
            $vals[] = (string) $m[1];
            continue;
        }
    }

    if ($vals === []) {
        return [];
    }

    return array_count_values($vals);
}

function leGetDrawsAgoForValue($db, string $dbCol, string $position, string $value, string $gId, int $drawRange): string
{
    $q = $db->getQuery(true)
        ->select('MAX(' . $db->quoteName('draw_date') . ')')
        ->from($db->quoteName($dbCol))
        ->where($db->quoteName('game_id') . ' = :gid')
        ->where($db->quoteName($position) . ' = :val');

    $q->bind(':gid', $gId, ParameterType::STRING);
    $q->bind(':val', $value, ParameterType::STRING);

    $db->setQuery($q);
    $lastDate = (string) $db->loadResult();

    if ($lastDate === '') {
        return 'Not in last ' . (int) $drawRange . ' dr.';
    }

    $q = $db->getQuery(true)
        ->select('COUNT(*)')
        ->from($db->quoteName($dbCol))
        ->where($db->quoteName('game_id') . ' = :gid')
        ->where($db->quoteName('draw_date') . ' > ' . $db->quote($lastDate));

    $q->bind(':gid', $gId, ParameterType::STRING);
    $db->setQuery($q);

    return (string) ((int) $db->loadResult()) . ' drws ago';
}

function leGetOverallComboFrequenciesPick3($db, string $dbCol, int $drawRange, string $gId, bool $includeExtra): array
{
    $cols = ['first', 'second', 'third'];
    if ($includeExtra) {
        $cols[] = 'fourth';
    }

    $q = $db->getQuery(true)
        ->select($db->quoteName($cols))
        ->from($db->quoteName($dbCol))
        ->where($db->quoteName('game_id') . ' = :gid')
        ->order($db->quoteName('draw_date') . ' DESC')
        ->setLimit($drawRange);

    $q->bind(':gid', $gId, ParameterType::STRING);
    $db->setQuery($q);

    $rows = $db->loadObjectList();
    if (!is_array($rows)) {
        return [];
    }

    $freq = [];
    foreach ($rows as $r) {
        $combo = (string) $r->first . (string) $r->second . (string) $r->third;
        if ($includeExtra) {
            $combo .= (string) $r->fourth;
        }
        if (!isset($freq[$combo])) {
            $freq[$combo] = 0;
        }
        $freq[$combo]++;
    }

    arsort($freq);
    return $freq;
}

<script>
(function(){
  function ready(fn){
    if (document.readyState === "complete" || document.readyState === "interactive") {
      setTimeout(fn, 0);
    } else {
      document.addEventListener("DOMContentLoaded", fn);
    }
  }

  ready(function(){
    var select = document.getElementById("heatmapSort");
    if (select) {
      select.addEventListener("change", function(){
        var mode = this.value;
        var grids = document.querySelectorAll(".digit-grid[data-sortgrid='1']");

        for (var i = 0; i < grids.length; i++) {
          var grid = grids[i];
          var cells = Array.prototype.slice.call(grid.querySelectorAll(".digit-cell"));

          cells.sort(function(a, b){
            var af = parseInt(a.getAttribute("data-freq"), 10) || 0;
            var bf = parseInt(b.getAttribute("data-freq"), 10) || 0;

            if (mode === "high") { return bf - af; }
            if (mode === "low")  { return af - bf; }

            var ad = parseInt(a.getAttribute("data-digit"), 10) || 0;
            var bd = parseInt(b.getAttribute("data-digit"), 10) || 0;
            return ad - bd;
          });

          for (var c = 0; c < cells.length; c++) {
            grid.appendChild(cells[c]);
          }
        }
      });
    }

    function initTable(sel, dir){
      if (!window.jQuery || !window.jQuery.fn || !window.jQuery.fn.DataTable) { return; }
      if (window.jQuery(sel).length === 0) { return; }
      if (window.jQuery(sel).hasClass("dataTable")) { return; }

      window.jQuery(sel).DataTable({
        "order": [[1, dir]],
        "paging": false,
        "bLengthChange": false,
        "bFilter": false,
        "bInfo": false
      });
    }

    var tries = 0;
    var timer = setInterval(function(){
      tries++;

      initTable("#firstPositionTable", "desc");
      initTable("#secondPositionTable", "desc");
      initTable("#thirdPositionTable", "desc");
      initTable("#extraBallTable", "desc");
      initTable("#overallFrequencyTable", "desc");

      if ((window.jQuery && window.jQuery.fn && window.jQuery.fn.DataTable) || tries >= 20) {
        clearInterval(timer);
      }
    }, 100);
  });
})();
</script>
